## Overview

The `HeaderOverflowPanel` component uses the overflow icon in the header to
display a panel with navigation options. It can be used to provide quick access
to a set of related actions or links, or to display additional information or
settings for sm and md breakpoints.

```jsx
import {
  HeaderOverflowPanel,
  SideNavDivider,
  SideNavLabs,
  SideNavItemsLabs,
  SideNavMenuLabs,
  SideNavLinkLabs,
} from '@afframe/ui';
import {
  Settings,
  ChartCustom,
  Notification,
  Help,
  Logout,
  Launch,
} from '@afframe/ui/icons';

<HeaderOverflowPanel>
  <SideNavLabs
    isTreeview
    isFixedNav
    expanded
    isChildOfHeader={false}
    aria-label="Header navigation"
    headerOverflowPanel>
    <SideNavItemsLabs>
      <SideNavMenuLabs
        renderIcon={Settings}
        primary
        title="Settings"
        backButtonTitle="Back">
        <SideNavLinkLabs renderIcon={ChartCustom} href="#">
          Customize
          <Launch />
        </SideNavLinkLabs>
      </SideNavMenuLabs>
      <SideNavLinkLabs renderIcon={Notification} href="#">
        Notifications
        <Launch />
      </SideNavLinkLabs>
      <SideNavLinkLabs renderIcon={Help} href="#">
        Help
        <Launch />
      </SideNavLinkLabs>
      <SideNavLinkLabs renderIcon={Help} href="#">
        Custom action
        <Launch />
      </SideNavLinkLabs>
      <SideNavDivider />
      <SideNavLinkLabs renderIcon={Logout} href="#">
        Logout
      </SideNavLinkLabs>
    </SideNavItemsLabs>
  </SideNavLabs>
</HeaderOverflowPanel>;
```
